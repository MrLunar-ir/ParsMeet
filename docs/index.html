مثال کامل

import ParsMeet

bot = ParsMeet.Bot(token="توکن ربات شما")

@bot.on_message_all()
def handler(data):
    chat_id = data["chat_id"]
    text = data["text"]
    if text == "/start":
        btn = bot.create_callback("منو", "منو")
        bot.send_message(chat_id, "سلام خوش آمدید", reply_markup={"inline_keyboard": [[btn]]})
    elif text == "/help":
        bot.send_message(chat_id, "دستورات: /start و /help")

bot.run()
