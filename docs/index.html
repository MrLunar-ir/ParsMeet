مثال کامل

import ParsMeet
from ParsMeet import KeyboardBuilder

bot = ParsMeet.Bot(token="توکن شما")

@bot.on_message_all()
def handler(data):
    chat_id = data["chat_id"]
    text = data["text"]
    if text == "/start":
        keyboard = KeyboardBuilder.create([
            [("پشتیبانی", "support"), ("سوالات متداول", "faq")],
            [("ارتباط با ادمین", "admin")]
        ])
        bot.send_message(chat_id, "سلام! چطور می‌تونم کمکتون کنم؟", reply_markup=keyboard)
    elif text.startswith("سوال"):
        answer = bot.ask_ai(text)
        bot.send_message(chat_id, answer)

@bot.on_callback_query()
def callback(data):
    if data["data"] == "support":
        bot.send_message(data["chat_id"], "برای پشتیبانی پیام بفرستید.")
    elif data["data"] == "admin":
        bot.send_message(data["chat_id"], "ادمین در حال حاضر آنلاین است.")

bot.run()
